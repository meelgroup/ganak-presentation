\section{Model counting}

\begin{frame}{Propositional model counting}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \begin{block}{\small Counting (\#SAT)}
        \small How many solutions does a CNF have?\\
        $(x_1 \lor x_2) \land (\neg x_1 \lor x_3)$ has 4.\\[2pt]
        One solution: NP. All of them: \textbf{\#P}.
      \end{block}
      \begin{block}{\small Weighted counting}
        \small Each literal has a weight; a solution's weight is the
        \textbf{product} of its literals' weights; the count is the
        \textbf{sum} over the solutions. Weights: rationals, complex
        numbers, polynomials\ldots
      \end{block}
    \end{column}
    \begin{column}{0.44\textwidth}
      \begin{block}{\small Projected counting}
        \small Count only distinct assignments to a \emph{projection set} $P$:
        \vspace{-1mm}
        \begin{center}\footnotesize
        \begin{tabular}{@{}c|c@{}}
          $P$ & other vars \\ \midrule
          \texttt{0001} & \texttt{010111} \\
          \texttt{0001} & \texttt{101010} \\
          \texttt{1011} & \texttt{111010} \\
        \end{tabular}
        \end{center}
        \vspace{-1mm}
        3 solutions, but only \textbf{2} over $P$.
      \end{block}
      \small
      Uses: probabilistic inference, quantitative verification
      (information leakage), reliability, product-line configuration,
      quantum circuit simulation.
    \end{column}
  \end{columns}
\end{frame}

\section{Search}

\begin{frame}{Probabilistic component caching}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \small
      \begin{itemize}
        \item A component = its unassigned variables + its
          unsatisfied clauses. Storing it exactly is \textbf{large}:
          hundreds of variables and clause IDs.
        \item \ganak{} stores only a \textbf{128-bit hash} of it,
          seeded from \texttt{--seed}
        \item Smaller entries $\Rightarrow$ many more components cached,
          and faster lookups
        \item The price: a collision gives a wrong count, with
          probability $\lesssim n^2 / 2^{128}$ over $n$ stored components
      \end{itemize}
    \end{column}
    \begin{column}{0.43\textwidth}
      \begin{block}{\small Probabilistically exact}
        \small The count is exact unless a hash collision happened.
        Rerun with another \texttt{--seed} to make it even less likely.
      \end{block}
      \scriptsize Introduced in \ganak{} (Sharma, Roy, Soos, Meel, IJCAI'19);
      entries are evicted by last use once \texttt{--maxcache} is reached.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Enhanced residual formula processing}
  Lift the SAT solver's techniques into the counter's SAT mode:
  \medskip
  \begin{itemize}
    \item \textbf{VSIDS scoring}: counters branch with VSADS\footnote{\scriptsize Sang et al., 2005},
      which is good for splitting into components. For finding \emph{one}
      solution, VSIDS is cheaper to compute and better.
    \item \textbf{Restarts} (Luby\footnote{\scriptsize Luby et al., 1993}):
      we only need one solution, so restarting the SAT search is safe.
    \item \textbf{Polarity caching}\footnote{\scriptsize Pipatsrisawat \& Darwiche, 2007}:
      remember the last polarity, so restarts do not throw the work away.
  \end{itemize}
\end{frame}

\begin{frame}{Dual independent set}
  \begin{definition}[Dual independent set]
    For a formula $F$ over $V$ with projection set $\mathcal{P}$:
    \begin{enumerate}
      \item \textbf{S-set} $S\subseteq \mathcal{P}$: once all of $S$ is
        decided, we may switch to SAT mode.
      \item \textbf{D-set} $D\supseteq \mathcal{P}$: the variables we may
        branch on.
    \end{enumerate}
    Traditionally $D = S \subseteq \mathcal{P}$. We aim for
    $S \subseteq \mathcal{P} \subseteq D$: a \emph{small} $S$ to reach SAT
    mode early, a \emph{large} $D$ to split into components well.
  \end{definition}
  \vspace{-3mm}
  \begin{center}

\begin{frame}{Computing a maximal decision set}
  \small
  \begin{algorithmic}[1]
    \State $D \gets \mathcal{P}$, \quad $G \gets \textsc{ExtractGates}(F)$
    \Procedure{SyntacticExpansion}{$V$}
      \State $\text{changed} \gets V$
      \While{changed is not empty}
        \State $v \gets \text{changed}.pop()$
        \For{gate $g \in G$ with $v$ as input} if all inputs are in $D$, add $g$'s output to $D$
        \EndFor
      \EndWhile
    \EndProcedure
    \Procedure{SemanticExpansion}{}
      \For{$v \notin D$}
        \If{$\textsc{ValidateDecisionVar}(D \cup \{v\}, F)$} \Comment{SAT call}
          \State $D \gets D \cup \{v\}$; \ \textsc{SyntacticExpansion}($\{v\}$)
        \EndIf
      \EndFor
    \EndProcedure
  \end{algorithmic}
  \medskip
  Run $\textsc{SyntacticExpansion}(\mathcal{P})$, then $\textsc{SemanticExpansion}$.
\end{frame}

\begin{frame}{Chronological backtracking}
  \begin{columns}[c]
    \begin{column}{0.45\textwidth}
      \small
      \begin{itemize}
        \item Backjumping throws away counted solutions, so counters
          backtrack one level, and drop the learnt clause if it would
          break the propagation invariants
        \item \textbf{ChronoBT} relaxes the invariants: implication levels
          may be out of order on the trail
        \item So we can \textbf{keep} the learnt clause, and force the
          literal it implies
        \item Fuzzed with \texttt{SharpVelvet} (Latour et al.)
      \end{itemize}
    \end{column}
    \begin{column}{0.55\textwidth}
      \centering
      \includegraphics[width=0.95\linewidth]{chronoBT.png}\\
      \scriptsize Figure from Nadel \& Ryvchin, \emph{Chronological Backtracking}, SAT 2018
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{ChronoBT needs compensating weights}

\begin{frame}{\new Dynamic blocked clause elimination}

      \small
      \begin{itemize}
        \item $C$ is \textbf{blocked} on $l \in C$ if every resolvent on
          $l$ is a tautology. Removing it keeps the count projected
          away from $\mathrm{var}(l)$
        \item \textbf{Dynamic}\footnote{\scriptsize Lagniez, Marquis, Biere, SAT'24}:
          under the current assignment satisfied clauses vanish, so
          more clauses become blocked. Ignore them when computing
          \textbf{components} $\Rightarrow$ finer splits, more cache hits
        \item Sound in \ganak{} only on variables we never branch on:
          not in the independent \emph{or} the optional set
        \item Incremental, undone in groups; ChronoBT can unassign
          a variable while later ones stay set, so those are re-processed
      \end{itemize}
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{exampleblock}{\small 1600 MCC'23+'24 instances}
        \small
        \begin{itemize}
          \item \textbf{+11} solved, $-1$ lost:\\ 1210 $\to$ \textbf{1220}
          \item PMC 282$\to$288, PWMC 336$\to$341
          \item PAR2 1951 $\to$ 1899
          \item projected, $>$10\,s: decisions $\times$0.42, time
            $\times$0.87 (geo.\ mean)
          \item 25 instances $\ge$2$\times$ faster, none $\ge$2$\times$ slower
        \end{itemize}
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\section{Tree decomposition}

\begin{frame}{\new Tree decomposition guides branching}
  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      \small
      Build a TD of the primal graph (FlowCutter, as in SharpSAT-TD\footnote{\scriptsize Korhonen \& J\"arvisalo, CP'21}).
      A variable's TD score is its bag's distance from the centroid bag:
      \[ \mathit{score}(v) = \tfrac{\mathit{act}(v)}{3} + w_{TD}\cdot \mathit{td}(v) + \tfrac{\mathit{freq}(v)}{25} \]
      What is new:
      \begin{itemize}
        \item \textbf{TD selection}: several candidate TDs (min-degree
          restarts on random relabellings, FlowCutter); the narrowest wins
        \item \textbf{Flat rule}: if width $\ge$ 50\% of the nodes, the
          graph is too dense for the TD to say anything: $w_{TD}=0$
        \item \textbf{Dense rule}: never trade width for a better split
      \end{itemize}
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{exampleblock}{\small 1600 MCC'23+'24 instances}
        \small
        \begin{itemize}
          \item selection: 1181 $\to$ 1198
          \item + rules: $\to$ \textbf{1205}\\ (+37, $-13$)
          \item PAR2 2084 $\to$ 1950
          \item commonly solved (1168): 285\,ks $\to$ \textbf{227\,ks} ($-20\%$)
        \end{itemize}
      \end{exampleblock}
      \scriptsize Width alone does not predict the work: the biggest
      wins came from TDs 6--12\% \emph{wider} but with far fewer levels.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{\new 2QBF cover-drop: removing what cannot matter}
  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      \small
      $\psi(X)$: the clauses over the projection set $X$ only;
      $C_i(X, Y_i)$: components over \emph{disjoint} non-projected $Y_i$.
      We count $\#_X\big(\psi \land \textstyle\bigwedge_i \exists Y_i.\, C_i\big)$.
      \smallskip

      For each $C_i$, solve the \textbf{2QBF}
      \ $\forall X.\; \psi(X) \rightarrow \exists Y_i.\, C_i(X, Y_i)$
      \begin{itemize}
        \item \textbf{true}: drop $C_i$ and its $Y_i$ entirely
        \item \textbf{false}, but only a few $X$-cubes have no $Y_i$:
          \textbf{replace} $C_i$ by clauses blocking those cubes
        \item weighted $Y_i$, or out of budget: keep
      \end{itemize}
      \scriptsize CEGAR over $X$; each counterexample is shortened to a
      cube by local search and SAT calls.
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{exampleblock}{\small 1600 MCC'23+'24 instances}
        \small
        \begin{itemize}
          \item changes \textbf{255} instances\\
            (223 projected)
          \item removes 2.4\,M clauses, 133\,k vars
          \item \textbf{+7} solved, $-0$ lost\\
            (PMC 278$\to$282, PWMC 333$\to$336)
          \item PAR2 1969 $\to$ 1950
        \end{itemize}
      \end{exampleblock}
      \scriptsize Cost: 61\,ks of extra preprocessing over the set, so
      time on the commonly solved ones goes up 4\%. (Off vs.\ on are
      two builds a few weeks apart.)
    \end{column}
  \end{columns}
\end{frame}
