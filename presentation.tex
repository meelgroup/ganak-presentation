\section{Model counting}

\begin{frame}{Propositional model counting}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \begin{block}{\small Counting (\#SAT)}
        \small How many solutions does a CNF have?\\
        $(x_1 \lor x_2) \land (\neg x_1 \lor x_3)$ has 4.\\[2pt]
        One solution: NP. Counting them: \textbf{\#P-complete}.\\[2pt]
        This talk: \emph{exact}. Approximate, with guarantees: ApproxMC.
      \end{block}
      \begin{block}{\small Weighted counting}
        \small Each literal has a weight; a solution's weight is the
        \textbf{product} of its literals' weights; the count is the
        \textbf{sum} over the solutions. Weights: rationals, complex
        numbers, polynomials\ldots
      \end{block}
    \end{column}
    \begin{column}{0.44\textwidth}
      \begin{block}{\small Projected counting}
        \small Count only distinct assignments to a \emph{projection set} $P$:
        \vspace{-1mm}
        \begin{center}\footnotesize
        \begin{tabular}{@{}c|c@{}}
          $P$ & other vars \\ \midrule
          \texttt{0001} & \texttt{010111} \\
          \texttt{0001} & \texttt{101010} \\
          \texttt{1011} & \texttt{111010} \\
        \end{tabular}
        \end{center}
        \vspace{-1mm}
        3 solutions, but only \textbf{2} over $P$.
      \end{block}
      \small
      Uses: probabilistic inference, quantitative verification
      (information leakage), reliability, product-line configuration,
      quantum circuit simulation.
    \end{column}
  \end{columns}
  \slideref{ApproxMC: Chakraborty, Meel, Vardi, CP'13.}
\end{frame}

\section{Preprocessing}

\begin{frame}{Dual independent set}
  \begin{definition}[Dual independent set]
    For a formula $F$ over $V$ with projection set $\mathcal{P}$:
    \begin{enumerate}
      \item \textbf{S-set} $S\subseteq \mathcal{P}$ \emph{defines} $\mathcal{P}$:
        once all of $S$ is decided, we may switch to SAT mode.
      \item \textbf{D-set} $D\supseteq \mathcal{P}$ is \emph{defined by} $\mathcal{P}$:
        we may branch on it, the projected count does not change.
    \end{enumerate}
    Traditionally $D = S \subseteq \mathcal{P}$. We aim for
    $S \subseteq \mathcal{P} \subseteq D$: a \emph{small} $S$ to reach SAT
    mode early, a \emph{large} $D$ to split into components well (CAV'25).
  \end{definition}
  \vspace{-5mm}

\begin{frame}{Preprocessing with Arjun}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{\small Small S-set + large D-set}
        \scriptsize
        Most variables are \emph{defined} by others (Padoa's theorem:
        two renamed copies cannot differ if the projection agrees).
        Arjun shrinks the projection to a small \textbf{S-set}, and grows
        it to a large \textbf{D-set} the counter \emph{may} branch on: gate
        outputs whose inputs are all in $D$, then Padoa checks.
      \end{block}
      \begin{block}{\small Simplified CNF}
        \scriptsize
        Fewer variables, fewer and shorter clauses. Every step must be
        \textbf{count-preserving} for the projection. BVE and
        equivalent-literal merging do most of the structural work.
      \end{block}
      \scriptsize
      Example \texttt{mc2023\_track3\_079}: 26\,351 vars $\to$ 420,
      112\,951 clauses $\to$ 2\,504, projection 75 $\to$ 12, in 8.5\,s.
    \end{column}
    \begin{column}{0.50\textwidth}
      \centering
      \begin{tikzpicture}[node distance=2.2mm, every node/.append style={text width=58mm}]

\section{Search}

\begin{frame}{The search: branching and component decomposition}
  \begin{columns}[c]
    \begin{column}{0.40\textwidth}
      \small
      \begin{itemize}
        \item Branch on a variable: the count is the \textbf{sum} of the two branches
        \item The residual formula falls apart into variable-disjoint
          \textbf{components}: the count is their \textbf{product}
        \item Count each component once, then \textbf{cache} it:
          the same component shows up again and again
        \item Conflicts: learn clauses, as in CDCL
      \end{itemize}
    \end{column}
    \begin{column}{0.58\textwidth}

\begin{frame}{The main loop}
  \centering
  \includegraphics[width=\linewidth]{figs/mermaid/main-loop.pdf}

  \bigskip
  \raggedright\small
  \begin{itemize}
    \item \textbf{Descend}: look the component up in the cache; on a miss, decide a
      variable in it and propagate. The next iteration cuts the result into components
    \item \textbf{Ascend}: add and multiply the sub-counts back up the stack
    \item We count on the way \emph{up}: much cheaper, e.g.\ a conflict
      means there is nothing to count
  \end{itemize}

\begin{frame}{Descending: decide, propagate, analyse conflicts}
  \centering
  \includegraphics[width=\linewidth]{figs/mermaid/descend.pdf}

  \bigskip
  \raggedright\small
  \begin{itemize}
    \item This is the CDCL part, very close to a SAT solver
    \item Counting-specific: branch only on variables of the \emph{current
      component}, and a conflict sets the count of its branch to 0
  \end{itemize}

\begin{frame}{Ascending: components, cache, count arithmetic}
  \centering
  \includegraphics[width=\linewidth]{figs/mermaid/ascend.pdf}

  \bigskip
  \raggedright\small
  \begin{itemize}
    \item Independent components \textbf{multiply}, the two values of a
      decision variable \textbf{add}
    \item Every component count is cached, so it is not searched again until evicted
    \item The same $+$ and $\times$ for every field: the engine never
      knows which one it counts in
  \end{itemize}
\end{frame}

\begin{frame}{Probabilistic component caching}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \small
      \begin{itemize}
        \item A component = its unassigned variables + its
          unsatisfied clauses. Storing it exactly is \textbf{large}:
          hundreds of variables and clause IDs.
        \item \ganak{} stores only a \textbf{128-bit hash} of it,
          seeded from \texttt{--seed}
        \item Smaller entries $\Rightarrow$ many more components cached,
          and faster lookups
        \item The price: a collision gives a wrong count, with
          probability $\approx n^2 / 2^{128}$ over $n$ stored components,
          if the hash behaves like a random function
      \end{itemize}
    \end{column}
    \begin{column}{0.43\textwidth}
      \begin{block}{\small Probabilistically exact}
        \small The count is exact unless a hash collision happened.
        Rerun with another \texttt{--seed} for an independent check.
      \end{block}
      \scriptsize Introduced in \ganak{} (Sharma, Roy, Soos, Meel, IJCAI'19)
      with a universal hash family and a formal bound; today MurmurHash3,
      128 bits. Entries are evicted by last use once \texttt{--maxcache} is reached.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{S-set and calling a SAT solver}
  \begin{columns}[c]
    \begin{column}{0.38\textwidth}
      \small
      Once all variables of the \textbf{S-set} $S$ (an independent support
      of the projection) are decided, there is \textbf{at most one}
      solution over $\mathcal{P}$ left
      \bigskip

      So call a \textbf{SAT solver} on the residual formula: it answers
      1 or 0. No more branching, components or cache below that point
      \bigskip

      $S=\{x_1, x_2\}$. Exploited before by \gpmc{}.
    \end{column}

\begin{frame}{Enhanced residual formula processing}
  Lift the SAT solver's techniques into the counter's SAT mode:
  \medskip
  \begin{itemize}
    \item \textbf{VSIDS scoring}: counters branch with VSADS\footnote{\scriptsize Sang et al., 2005},
      which is good for splitting into components. For finding \emph{one}
      solution, VSIDS is cheaper to compute and better.
    \item \textbf{Restarts} (Luby\footnote{\scriptsize Luby et al., 1993}):
      we only need one solution, so restarting the SAT search is safe.
    \item \textbf{Polarity caching}\footnote{\scriptsize Pipatsrisawat \& Darwiche, 2007}:
      remember the last polarity, so restarts do not throw the work away.
  \end{itemize}
\end{frame}

    \node[anchor=west, font=\scriptsize, align=left, text=hl] at (3.0,-1.4)
      {levels 2 and 3 undone:\\ the work at level 2 is lost};

    \tdec{0,-2.8}{x_1}{1}{stageA} \tlit{1,-2.8}{x_2}{1}{stageA}
    \tdec{2.2,-2.8}{x_3}{2}{stageB} \tlit{3.2,-2.8}{x_4}{2}{stageB}
    \tlit{4.4,-2.8}{x_6}{1}{stageA}
    \draw[hl, very thick, rounded corners=2pt] (3.96,-3.17) rectangle (4.84,-2.43);
    \node[anchor=west, font=\scriptsize, align=left, text=stageB!70!black] at (5.1,-2.8)
      {only level 3 undone; $x_6$ keeps level 1\\ but sits above level 2: \textbf{out of order}};
  \end{tikzpicture}

  \medskip
  \raggedright\small
  \begin{itemize}
    \item \textbf{Backjumping throws away counted solutions}, so counters
      backtrack one level, and drop the learnt clause if it would
      break the propagation invariants
    \item \textbf{ChronoBT}\footnote{\scriptsize Nadel \& Ryvchin,
      \emph{Chronological Backtracking}, SAT 2018. In counting:
      M\"ohle \& Biere, GCAI 2019} relaxes the invariants:
      implication levels may be \textbf{out of order} on the trail
    \item \textbf{Keep} the learnt clause, and force the
      literal it implies, \textbf{without} undoing the levels in between
  \end{itemize}
\end{frame}

\begin{frame}{ChronoBT needs compensating weights}
  \begin{columns}[c]
    \begin{column}{0.47\textwidth}
      \small
      \begin{enumerate}
        \item Nodes 1 and 2 are counted. $x_5$ holds in all their models,
          so $w(x_5)$ is already in their counts
        \item At level 3: conflict, learn the \textbf{unit} $x_5$. ChronoBT
          undoes only level 3, and puts $x_5$ at \textbf{level 0}, above every
          decision
        \item Level-0 weights multiply the \textbf{whole} count at the end:
          $w(x_5)$ would be counted twice in nodes 1 and 2
        \item So \textbf{divide} $w(x_5)$ out of every finished branch on the
          path: nodes 1 and 2
      \end{enumerate}
    \end{column}
    \begin{column}{0.51\textwidth}
      \centering
      \begin{tikzpicture}[x=1cm, y=1cm]
        \node[lev]   (l0) at (0,0)      {lev 1};
        \node[lev]   (l1) at (1.6,-1.4) {lev 2};
        \node[confl] (l3) at (3.2,-2.8) {lev 3 \Lightning};
        \node[comp]  (n1) at (-1.3,-1.4) {node 1};
        \node[comp]  (n2) at (0.3,-2.8)  {node 2};
        \foreach \a/\b/\l in {l0/n1/{x_1=0}, l1/n2/{x_2=0}}
          \draw[fl] (\a) -- (\b) node[lbl, midway, above left] {$\l$};
        \foreach \a/\b/\l in {l0/l1/{x_1=1}, l1/l3/{x_2=1}}
          \draw[fl] (\a) -- (\b) node[lbl, midway, above right] {$\l$};
        \node[lbl, font=\scriptsize\bfseries, text=hl, below=1mm of n1] {$\div\, w(x_5)$};
        \node[lbl, font=\scriptsize\bfseries, text=hl, below=1mm of n2] {$\div\, w(x_5)$};
        \node[lbl, align=center, below=1mm of l3] {unit $x_5$,\\ goes to level 0};
      \end{tikzpicture}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Compensating across components}
  \begin{columns}[c]
    \begin{column}{0.40\textwidth}
      \small
      \begin{enumerate}
        \item $x_4$ is in the components of nodes 1--4, but not 5. Yet at 5
          we learn $\neg x_4$ at level 0
        \item How? Components are computed from the \emph{irredundant}
          clauses only. Learnt clauses can \textbf{connect} components
        \item So conflicts can be over variables \emph{outside} the
          component examined
        \item So divide $w(\neg x_4)$ out of the finished branch $x_1=0$
          (nodes 1+2) \emph{and} the finished sibling component (nodes 3+4)
      \end{enumerate}
    \end{column}
    \begin{column}{0.58\textwidth}
      \centering
      \begin{tikzpicture}[x=1cm, y=1cm]
        \node[lev] (l0) at (0,0) {lev 1};

\begin{frame}{Dynamic blocked clause elimination}
  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      \small
      \begin{itemize}
        \item $C$ is \textbf{blocked} on $l \in C$ if every resolvent on
          $l$ is a tautology. Removing it keeps the count projected
          away from $\mathrm{var}(l)$
        \item \textbf{Dynamic}\footnote{\scriptsize Lagniez, Marquis, Biere, SAT'24. Blocked clauses: Kullmann, 1999}:
          under the current assignment satisfied clauses vanish, so
          more clauses become blocked. Ignore them when computing
          \textbf{components} $\Rightarrow$ finer splits, more cache hits
        \item Sound in \ganak{} only on variables we never branch on:
          outside the D-set
        \item Incremental, undone in groups; ChronoBT can unassign
          a variable while later ones stay set, so those are re-processed
      \end{itemize}
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{exampleblock}{\small 1600 MCC'23+'24 instances}
        \small
        \begin{itemize}
          \item \textbf{+11} solved, $-1$ lost:\\ 1210 $\to$ \textbf{1220}
          \item PMC 282$\to$288, PWMC 336$\to$341
          \item PAR2 1951 $\to$ 1899
          \item projected, $>$10\,s: decisions $\times$0.42, time
            $\times$0.87 (geo.\ mean)
          \item 25 instances $\ge$2$\times$ faster, none $\ge$2$\times$ slower
        \end{itemize}
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\section{Tree decomposition}

\begin{frame}{Tree decomposition guides branching}
  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      \small
      Build a TD of the primal graph (FlowCutter, as in SharpSAT-TD\footnote{\scriptsize Korhonen \& J\"arvisalo, CP'21}).
      The closer a variable's bag is to the centroid bag, the higher its TD score:
      \[ \mathit{score}(v) = \tfrac{\mathit{act}(v)}{3} + w_{TD}\cdot \mathit{td}(v) + \tfrac{\mathit{freq}(v)}{25} \]
      Recent changes:
      \begin{itemize}
        \item \textbf{TD selection}: several candidate TDs (min-degree
          restarts on random relabellings, FlowCutter); the narrowest wins
        \item \textbf{Flat rule}: if width $\ge$ 50\% of the nodes, the
          graph is too dense for the TD to say anything: $w_{TD}=0$
        \item \textbf{Dense rule}: never trade width for a better split
      \end{itemize}
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{exampleblock}{\small 1600 MCC'23+'24 instances}
        \small
        \begin{itemize}
          \item selection: 1181 $\to$ 1198
          \item + rules: $\to$ \textbf{1205}\\ (+37, $-13$)
          \item PAR2 2084 $\to$ 1950
          \item commonly solved (1168): 285\,ks $\to$ \textbf{227\,ks} ($-20\%$)
        \end{itemize}
      \end{exampleblock}
      \scriptsize Width alone does not predict the work: TDs of the same
      width can differ 2.4$\times$ in decisions.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Counting over any field}

\begin{frame}{2QBF cover-drop: removing what cannot matter}
  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      \small
      $\psi(X)$: the clauses over the projection set $X$ only;
      $C_i(X, Y_i)$: components over \emph{disjoint} non-projected $Y_i$.
      We count $\#_X\big(\psi \land \textstyle\bigwedge_i \exists Y_i.\, C_i\big)$.
      \smallskip

      For each $C_i$, solve the \textbf{2QBF}
      \ $\forall X.\; \psi(X) \rightarrow \exists Y_i.\, C_i(X, Y_i)$
      \begin{itemize}
        \item \textbf{true}: drop $C_i$ and its $Y_i$ entirely
        \item \textbf{false}, but only a few $X$-cubes have no $Y_i$:
          \textbf{replace} $C_i$ by clauses blocking those cubes
        \item weighted $Y_i$, or out of budget: keep
      \end{itemize}
      \scriptsize CEGAR over $X$; each counterexample is shortened to a
      cube by local search and SAT calls.
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{exampleblock}{\small 1600 MCC'23+'24 instances}
        \small
        \begin{itemize}
          \item changes \textbf{255} instances\\
            (223 projected)
          \item removes 2.4\,M clauses, 133\,k vars
          \item \textbf{+7} solved, $-0$ lost\\
            (PMC 278$\to$282, PWMC 333$\to$336)
          \item PAR2 1969 $\to$ 1950
        \end{itemize}
      \end{exampleblock}
      \scriptsize Cost: 61\,ks of extra preprocessing over the set, so
      time on the commonly solved ones goes up 4\%.
    \end{column}
  \end{columns}
\end{frame}
